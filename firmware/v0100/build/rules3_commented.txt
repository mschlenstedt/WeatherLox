// Version of Rules 3 for the automatic update (Rules 1) - must match the version in the first line
On version3 Do
  Let,rv3,10103
Endon

// Calculate rainrate
On input_1#All Do
  If %eventvalue1%=1
    Let,rain_event,%v_rain_event%+%v_rainbucket%
    Let,rain_hourly,%v_rain_hourly%+%v_rainbucket%
    Let,rain_daily,%v_rain_daily%+%v_rainbucket%
    Let,rain_weekly,%v_rain_weekly%+%v_rainbucket%
    Let,rain_monthly,%v_rain_monthly%+%v_rainbucket%
    Let,rain_yearly,%v_rain_yearly%+%v_rainbucket%
    TaskValueSet data_5,rain_event,%v_rain_event%
    TaskValueSet data_5,rain_hourly,%v_rain_hourly%
    TaskValueSet data_6,rain_daily,%v_rain_daily%
    TaskValueSet data_6,rain_weekly,%v_rain_weekly%
    TaskValueSet data_6,rain_monthly,%v_rain_monthly%
    TaskValueSet data_6,rain_yearly,%v_rain_yearly%  
    If %unixtime%<(%v_rainratelast%+600)
      TaskValueSet data_5,rainrate,%v_rainbucket%/(%unixtime%-%v_rainratelast%)*3600 // %v_rainbucket% mm per Pulse, converted to mm/h
    Endif
    Let,rainratelast,%unixtime%
  Endif
Endon

// Calculate rain status (and rainrate < 1 mm/h)
On input_2#All Do
  TaskValueSet data_5,rainstate,%eventvalue1%
  If %unixtime%>(%v_rainratelast%+600) And [data_5#rainrate]=0 // This maybe is the first apperance - so reset timer
    Let,rainratelast,%unixtime%
  Endif
  If %eventvalue1%=1 And [data_5#rainrate]=0 // If we detect rain here but the bucket counter is still zero, set 0.5 mm/h here (minimum of bucket counter is 1.2 mm/h)
    TaskValueSet data_5,rainrate,0.5
  Endif
Endon

// BMx280 Sensor (BME280: temperature, humidity, pressure - BMP280: temperature, pressure, humidity is always 0)
// Pressure only if there is no BMP58x (more accurate), temperature/humidity only if there is no SHT3x
On bmx280_1#All Do
  If %unixtime%-%v_bmp58xlast%>120 // No value from a BMP58x for 2 minutes (or none connected)
    Event,calcpressure=%eventvalue3%
  Endif
  If [sht3x_1#settings.enabled]=0 And %eventvalue2%>0 // No SHT3x available and humidity from a BME280
    Event,calctemps=%eventvalue1%,%eventvalue2%
  Endif
Endon

// BMP58x Sensor via Generic I2C (P180), Output Data Type Temp / Baro: %eventvalue1% = temperature (°C), %eventvalue2% = pressure (hPa)
// Has priority over a BMx280 for the pressure. Task interval must be 60 s or less.
On bmp58x_1#All Do
  Let,bmp58xlast,%unixtime%
  Event,calcpressure=%eventvalue2%
Endon

// Absolute and relative (sea level) air pressure: %eventvalue1% = absolute pressure (hPa)
On calcpressure Do
  TaskValueSet data_3,pressure_abs,%eventvalue1%
  // Sea level pressure (rel) as QFF like the DWD: Leitfaden Nr. 6 "Flugmeteorologie", formula 7.3, page 25,
  // https://www.dwd.de/DE/leistungen/pbfb_verlag_leitfaeden/pdf_einzelbaende/leitfaden6_pdf.pdf?__blob=publicationFile&v=3
  // QFF = p * exp((H*gn/RL) / (T + e*Ch + a*H/2)) - gn 9.80665, RL 287.058, a 0.0065 K/m, Ch 0.12 K/hPa,
  // H = station height (m), T = temperature (K), e = vapour pressure (hPa, vp from calctemps is in kPa)
  If %v_stationheight%>0
    Let,qff,(%v_stationheight%*9.80665/287.058)/(273.15+[data_1#temperature]+0.12*%v_vp%*10+0.0065*%v_stationheight%/2)
    TaskValueSet data_3,pressure_rel,%eventvalue1%*(exp(%v_qff%)) // ESPEasy: a function takes everything behind it - always wrap it in brackets
  Else
    TaskValueSet data_3,pressure_rel,%eventvalue1%
  Endif
Endon

// SHT3x Sensor
On sht3x_1#All Do
  Event,calctemps=%eventvalue1%,%eventvalue2%
Endon

// Calculate Temperature, Humidity, Humidity abs, Vapour Pressure Deficit, Dewpoint, Cloud Base, Windchill, Heat Index, Feelslike
// %eventvalue1% = Temperature (°C), %eventvalue2% = rel. Humidity (%)
// Formulas and limits as in FOSHKplugin
On calctemps Do
  TaskValueSet data_1,temperature,%eventvalue1%
  TaskValueSet data_1,humidity,%eventvalue2%
  // Saturation vapour pressure and actual vapour pressure in kPa (Magnus formula, coefficients Alduchov-Eskridge)
  Let,svp,0.61094*exp((17.625*%eventvalue1%)/(%eventvalue1%+243.04))
  Let,vp,%v_svp%*%eventvalue2%/100
  // Absolute humidity in g/m³: 216.7 * vp(hPa) / T(K)
  TaskValueSet data_2,humidity_abs,2167*%v_vp%/(%eventvalue1%+273.15)
  TaskValueSet data_2,vapourpressuredeficit,%v_svp%-%v_vp%
  // Dewpoint (Magnus formula, coefficients Alduchov-Eskridge)
  Let,magnus,(ln(%eventvalue2%/100))+(17.625*%eventvalue1%)/(%eventvalue1%+243.04) // ESPEasy: a function takes everything behind it - always wrap it in brackets
  Let,dewpoint,243.04*%v_magnus%/(17.625-%v_magnus%)
  TaskValueSet data_1,dewpoint,%v_dewpoint%
  // Height of Cloud Base above ground, Henningsche Formel, see https://www.dwd.de/DE/leistungen/pbfb_verlag_vub/pdf_einzelbaende/vub_16_gesamt_pdf.pdf?__blob=publicationFile&v=5
  TaskValueSet data_3,cloudbase,(%eventvalue1%-%v_dewpoint%)*125
  // Windchill, https://www.windinfo.eu/windchill/ - Temp up to 10°C and Wind from 4.828 km/h (3 mph)
  If %eventvalue1%<=10 And [data_7#windspeed_avg2m]>=4.828
    TaskValueSet data_2,windchill,13.12+(0.6215*%eventvalue1%)-(11.37*[data_7#windspeed_avg2m]^0.16)+(0.3965*%eventvalue1%*[data_7#windspeed_avg2m]^0.16)
  Else
    TaskValueSet data_2,windchill,%eventvalue1%
  Endif
  // Heat Index, https://www.wpc.ncep.noaa.gov/html/heatindex_equation.shtml
  Let,tf,%c_c2f%(%eventvalue1%)
  Let,heatindex,0.5*(%v_tf%+61+((%v_tf%-68)*1.2)+(%eventvalue2%*0.094)) // Simple equation, always first
  If %v_heatindex%>=80 // Complex equation (Rothfusz)
    Let,heatindex,-42.379+2.04901523*%v_tf%+10.14333127*%eventvalue2%-0.22475541*%v_tf%*%eventvalue2%-0.00683783*%v_tf%^2-0.05481717*%eventvalue2%^2+0.00122874*%v_tf%^2*%eventvalue2%+0.00085282*%v_tf%*%eventvalue2%^2-0.00000199*%v_tf%^2*%eventvalue2%^2
    If %eventvalue2%<13
      If %v_tf%>=80 And %v_tf%<=112 // Low humidity adjustment
        Let,heatindex,%v_heatindex%-((13-%eventvalue2%)/4)*sqrt((17-abs(%v_tf%-95))/17)
      Endif
    Elseif %eventvalue2%>85
      If %v_tf%>=80 And %v_tf%<=87 // High humidity adjustment
        Let,heatindex,%v_heatindex%+((%eventvalue2%-85)/10)*((87-%v_tf%)/5)
      Endif
    Endif
  Endif
  TaskValueSet data_2,heatindex,(%v_heatindex%-32)*0.55555555 // Convert to C
  // Feelslike as in FOSHKplugin / Ecowitt: Windchill up to 10°C with wind, Heat Index from 80°F (26.7°C), otherwise temperature. NOT similar to DWD.
  If %eventvalue1%<=10 And [data_7#windspeed_avg2m]>4.828
    TaskValueSet data_1,feelslike,[data_2#windchill]
  Elseif %v_tf%>=80
    TaskValueSet data_1,feelslike,[data_2#heatindex]
  Else
    TaskValueSet data_1,feelslike,%eventvalue1%
  Endif
Endon

// Lightning - AS3935 Sensor (ESPEasy P169), event every task interval:
// %eventvalue1% = distance near (km), %eventvalue2% = distance far (km), %eventvalue3% = strikes since last read, %eventvalue4% = strikes since boot
On as3935_1#All Do
  If %eventvalue3%>0
    Let,lightningdist,%eventvalue1% // Distance of the nearest strike: the smaller of both values
    If %eventvalue2%<%v_lightningdist%
      Let,lightningdist,%eventvalue2%
    Endif
    TaskValueSet data_15,lightning,%v_lightningdist%
    TaskValueSet data_15,lightning_time,%unixtime%
    Let,lightning_daily,%v_lightning_daily%+%eventvalue3%
    TaskValueSet data_15,lightning_daily,%v_lightning_daily%
  Endif
Endon

// 1-Wire temperature sensors - DS18B20 and compatible (ESPEasy P004, task ow_1, up to 4 sensors in series on one GPIO):
// %eventvalue1% ... %eventvalue4% = temperature 1-4 (°C). The task ow_1 is disabled by default and must be enabled manually.
On ow_1#All Do
  TaskValueSet data_14,temperature1,%eventvalue1%
  TaskValueSet data_14,temperature2,%eventvalue2%
  TaskValueSet data_14,temperature3,%eventvalue3%
  TaskValueSet data_14,temperature4,%eventvalue4%
Endon

// Calculate Illuminance
On veml7700_1#All Do
  TaskValueSet data_10,illuminance,%eventvalue1%
Endon

// Calculate Solar Radiation
On ina219_1#All Do
  If %eventvalue1%<0.0005 // sometimes slightly negativ or small values due to offset
      TaskValueSet data_10,solarradiation,0
  Else
    TaskValueSet data_10,solarradiation,%eventvalue1%*1792.1147 // (%eventvalue1%*1000/558)*1000 - Short circuit current (mA): 558 mA at 1000 W/m^2
  Endif
  Let,solarradiation_daily,(%v_solarradiation_daily%+([data_10#solarradiation]*[ina219_1#settings.interval]/3600)) // Wh/m² - one value per task interval (s)
  TaskValueSet data_11,solarradiation_daily,%v_solarradiation_daily%
  If [data_10#solarradiation]>%v_seuil% And [data_10#solarradiation]>%v_sunmin% // Like FOSHKplugin: above threshold seuil (Rules 4, calcsun; 0 below 3° sun elevation) and above sunmin
    TaskValueSet data_9,sunshine,1
    Let,sunduration_daily,(%v_sunduration_daily%+([ina219_1#settings.interval]/3600)) // h
    TaskValueSet data_11,sunduration_daily,%v_sunduration_daily%
  Else
    TaskValueSet data_9,sunshine,0
  Endif
Endon

 // Calculate Twilight, UVI, Windspeed, Windidr, Windgust, Windrun, Beaufort
On ads1115_1#All Do
  // Twilight
  Let,twilight,%eventvalue1%/0.0325 // /3.25*100
  TaskValueSet data_10,twilight,{constrain:%v_twilight%:0:100} // 3.3V (real: 3.25V) is absolute darkness = 100%
  
  // UVI
  TaskValueSet data_10,uvi,%eventvalue2%/0.1 // 1V = UVI 10. Steps in 0.1V
  
  // Wind Measures - Save
  TaskValueSet internal_1,int_random,%c_random%(0, 1) // Force the Statistics to be updated
  TaskValueSet internal_1,int_windspeed,mapc(%eventvalue3%:0.005:2.990:0:60)*3.6 // 2.99V maximum, 60 m/s
  TaskValueSet data_7,windspeed,[internal_1#int_windspeed]
  If %v_winddirsensor%=1
    TaskValueSet data_8,winddir,mapc(%eventvalue4%:0.005:2.809:0:315) // For 8-Point-Sensor
  Else
    TaskValueSet data_8,winddir,mapc(%eventvalue4%:0.005:2.99:0:359) // For 0-360° Sensor
  Endif
  TaskValueSet internal_1,int_winddir_x,cos_d([data_8#winddir])
  TaskValueSet internal_1,int_winddir_y,sin_d([data_8#winddir]) // ....and Run to Save the stats
  TaskRun internal_1
  // Wind Speed AVG and Max - avg24/avg120 assume a task interval of 5 s (24 values = 2 min, 120 values = 10 min)
  TaskValueSet data_7,windspeed_avg2m,[internal_1#int_windspeed.avg24]  // Average of windspeed - every 5 sec a value. 2 Min AVG
  TaskValueSet data_7,windspeed_avg10m,[internal_1#int_windspeed.avg120] // Average of windspeed - every 5 sec a value. 10 Min AVG
  TaskValueSet data_7,windgust_max2m,[internal_1#int_windspeed.max24] // Max of windspeed - every 5 sec a value. 2 Min MAX
  TaskValueSet data_8,windgust_max10m,[internal_1#int_windspeed.max120]  // Max of windspeed - every 5 sec a value. 10 Min MAX
  TaskValueSet data_9,beaufort,%c_ms2Bft%([data_7#windspeed_avg2m]/3.6)
  If [data_7#windspeed]>[data_9#windgust_maxdaily] // Setting daily maximum
    TaskValueSet data_9,windgust_maxdaily,[data_7#windspeed]
  Endif
  Let,windrun_daily,(%v_windrun_daily%+([internal_1#int_windspeed]*[ads1115_1#settings.interval]/3600)) // km - one value per task interval (s)
  TaskValueSet data_9,windrun_daily,%v_windrun_daily%
  // Winddirection AVG: vector mean like FOSHKplugin avgWind - every 5 sec a value, avg24 = 2 min, avg120 = 10 min
  // ESPEasy bug (mega-20260720): 2-argument functions (atan2_d, fmod) need brackets around the whole call and around
  // a calculated first argument, otherwise operators get lost: 360+atan2_d(-1:-1) = -135, fmod(5+5:3) = 5
  TaskValueSet data_8,winddir_avg2m,fmod((atan2_d([internal_1#int_winddir_y.avg24]:[internal_1#int_winddir_x.avg24])+360):360)
  TaskValueSet data_8,winddir_avg10m,fmod((atan2_d([internal_1#int_winddir_y.avg120]:[internal_1#int_winddir_x.avg120])+360):360)
Endon
