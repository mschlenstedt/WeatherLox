 // V1.0.2 - 17.01.2026

on WiFi#Connected Do
  // Some default vars
  LetStr,ecowittstationtype,"WeatherLox"
  LetStr,ecowittmodel,"WEATHERLOX_V1.0"
  LetStr,ecowittmac,{urlencode:"%mac%"}
  Let,rainratelast,0
  
  // Now we're enabling tasks for I2C sensors as they may have been disabled when not found
  Event,EnableSensors

  // Restore sum values from weatherlox webserver, currently there's no option in ESPEasy to save values
  // Only first time after cold reboot
  If %v_savesumvalues%>0 And %v_sumvaluesset%<1
    SendToHTTP,webservices.weatherlox.de,80,"getdata.php?mac={urlencode:"%mac%"}#json3"
  Endif
  Let,sumvaluesset,1
  // Start Timer for some quarterly tasks / events
  LoopTimerSetAndRun,1,900
  // Start Timer for some 5 seconds tasks / events
  LoopTimerSetAndRun,2,5
  // Start Timer for some 5 minutes tasks / events
  LoopTimerSetAndRun,3,300
  // Get station height from Open Meteo if not set here in the rules manually
  If %v_stationheight%=0
    SendToHTTP,api.open-meteo.com,80,"v1/elevation?latitude=%latitude%&longitude=%longitude%#json1"
  Endif
  // Get passkey for Ecowitt Format for our mac address if enabled - ESPEasy cannot calculate MD5 checksums
  If [data_1#settings.controller3.enabled]>0
    SendToHTTP,webservices.weatherlox.de,80,"ecowittpasskey.php?mac={urlencode:"%mac%"}#json2"
  Endif
Endon

// Enable tasks for I2C sensors - ESPEasy disables a task if the sensor is not found at startup
// Called at WiFi#Connected and every 5 minutes (Rules 4). Add new I2C sensors here.
On EnableSensors Do
  TaskEnable,ads1115_1
  TaskEnable,veml7700_1
  TaskEnable,ina219_1
  TaskEnable,bmx280_1
  TaskEnable,as3935_1
  TaskEnable,sht3x_1
Endon

// Parse data from Open Meteo
On OpenMeteo#current Do
    TaskValueSet data_13,precipitationprob,%eventvalue1%
    TaskValueSet data_13,visibility,%eventvalue2%/1000
    TaskValueSet data_13,evapotranspiration,%eventvalue3%
    TaskValueSet data_13,et0evapotranspiration,%eventvalue4%
    TaskValueSet data_12,weathercode,%eventvalue5% // https://www.nodc.noaa.gov/archive/arc0021/0002199/1.1/data/0-data/HTML/WMO-CODE/WMO4677.HTM
    TaskValueSet data_12,cloudcover,%eventvalue6%
Endon

// Restore sum values after power loss - only values from the current day / week / month / year
// %eventvalue1-9% = solarradiation_daily, sunduration_daily, windgust_maxdaily, windrun_daily, rain_daily, rain_weekly, rain_monthly, rain_yearly, lightning_daily
// %eventvalue10% = timestamp of the last save (unixtime UTC, set by the server), "null" if there is no saved data
On restoresums Do
  Let,savedts,0
  Let,savedts,%eventvalue10% // Fails for "null" and savedts stays 0 (a string comparison with null would be true)
  If %v_savedts%>0
    Let,daystart,%unixtime%-%syssec_d% // Today 0:00 local time
    Let,weekstart,%v_daystart%-((%sysweekday%+5)%7)*86400 // Monday 0:00 (%sysweekday%: 1=Sunday, 2=Monday)
    Let,monthstart,%v_daystart%-(%sysday%-1)*86400 // 1st of month 0:00
    Let,yearstart,(((%sysyear%-1970)*365+((%sysyear%-1969-((%sysyear%-1969)%4))/4))*86400)-%systzoffset_s% // 1st January 0:00
    If %v_savedts%>=%v_daystart%
      Let,solarradiation_daily,%eventvalue1%
      TaskValueSet data_11,solarradiation_daily,%eventvalue1%
      Let,sunduration_daily,%eventvalue2%
      TaskValueSet data_11,sunduration_daily,%eventvalue2%
      TaskValueSet data_9,windgust_maxdaily,%eventvalue3%
      Let,windrun_daily,%eventvalue4%
      TaskValueSet data_9,windrun_daily,%eventvalue4%
      Let,rain_daily,%eventvalue5%
      TaskValueSet data_6,rain_daily,%eventvalue5%
      Let,lightning_daily,%eventvalue9%
      TaskValueSet data_15,lightning_daily,%eventvalue9%
    Endif
    If %v_savedts%>=%v_weekstart%
      Let,rain_weekly,%eventvalue6%
      TaskValueSet data_6,rain_weekly,%eventvalue6%
    Endif
    If %v_savedts%>=%v_monthstart%
      Let,rain_monthly,%eventvalue7%
      TaskValueSet data_6,rain_monthly,%eventvalue7%
    Endif
    If %v_savedts%>=%v_yearstart%
      Let,rain_yearly,%eventvalue8%
      TaskValueSet data_6,rain_yearly,%eventvalue8%
    Endif
  Endif
Endon

// Parse data from JSON sources
On JsonReply* Do
  If %eventpar%=1 // Station height for pressure calculations
    Let,stationheight,%eventvalue1%
  Elseif %eventpar%=2 // Passkey for Ecowitt
    LetStr,ecowittpasskey,%eventvalue1%
  Elseif %eventpar%=3 // Saved sum values from server (order as in json.keys)
    Event,restoresums=%eventvalue1%,%eventvalue2%,%eventvalue3%,%eventvalue4%,%eventvalue5%,%eventvalue6%,%eventvalue7%,%eventvalue8%,%eventvalue9%,%eventvalue10%
  Endif
Endon
