// Version of Rules 2 for the automatic update (Rules 1) - must match the version in the first line
On version2 Do
  Let,rv2,10108
Endon

on WiFi#Connected Do
  // Some default vars
  LetStr,ecowittstationtype,"WeatherLox"
  LetStr,ecowittmodel,"WEATHERLOX_V1.0"
  LetStr,ecowittmac,{urlencode:"%mac%"}
  Let,rainratelast,0
  
  // Now we're enabling tasks for I2C sensors as they may have been disabled when not found
  Event,EnableSensors

  // Start Timer for some quarterly tasks / events
  LoopTimerSetAndRun,1,900
  // Start Timer for some 5 seconds tasks / events
  LoopTimerSetAndRun,2,5
  // Start Timer for some 5 minutes tasks / events - also requests station height, Ecowitt passkey and saved sum values (Rules 4)
  LoopTimerSetAndRun,3,300
Endon

  TaskEnable,veml7700_1
  TaskEnable,ina219_1
  TaskEnable,bmx280_1
  TaskEnable,bmp58x_1
  TaskEnable,as3935_1
  TaskEnable,sht3x_1

// Parse data from Open Meteo
// ESPEasy sets -256 for every value missing in the response (e.g. "503 service overloaded"): keep the last valid value
On OpenMeteo#current Do
  If %eventvalue1%!=-256
    TaskValueSet data_13,precipitationprob,%eventvalue1%
  Endif
  If %eventvalue2%!=-256
    TaskValueSet data_13,visibility,%eventvalue2%/1000
  Endif
  If %eventvalue3%!=-256
    TaskValueSet data_13,evapotranspiration,%eventvalue3%
  Endif
  If %eventvalue4%!=-256
    TaskValueSet data_13,et0evapotranspiration,%eventvalue4%
  Endif
  If %eventvalue5%!=-256
    TaskValueSet data_12,weathercode,%eventvalue5% // https://www.nodc.noaa.gov/archive/arc0021/0002199/1.1/data/0-data/HTML/WMO-CODE/WMO4677.HTM
  Endif
  If %eventvalue6%!=-256
    TaskValueSet data_12,cloudcover,%eventvalue6%
  Endif
Endon

// Block the tasks with sum values after a power loss (%bootcause%=1, RTC memory empty), so that no controller sends the sums as 0.
// TaskInit comes right at the start, before the first reading and before WiFi (the MQTT controller queues values also without a connection).
// While a task is blocked, [task#value] is empty and TaskValueSet is ignored - the sums keep counting in the variables (Rules 3).
// Released by releasesums as soon as the sums are valid (Rules 4, Timer 2).

On TaskInit#data_5 Do
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_5
  Endif
Endon

On TaskInit#data_6 Do
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_6
    Event,blockcontrollers
  Endif
Endon

// After a power loss, also switch off every controller that gets sum values: this clears its queue, which may already hold
// the first readings (0) of the sum tasks from before TaskInit. Switched on again by releasesums.
// Needs [task#settings.controllerN.enabled] to return the controller setting of the task (ESPEasy fix for issue #5668).
// Older firmware returns the task state (1) for every N, also for N=9 - then the controllers are not touched,
// because switching on a controller the user has switched off must never happen.
On blockcontrollers Do
  Let,ctrlfix,1
  If [data_1#settings.controller9.enabled]=1
    Let,ctrlfix,0
  Endif
  If %v_ctrlfix%=1
    Let,c1off,0
    Let,cs,0
    Let,cs,[data_5#settings.controller1.enabled]+[data_6#settings.controller1.enabled]+[data_9#settings.controller1.enabled]+[data_11#settings.controller1.enabled]+[data_15#settings.controller1.enabled]+[data_16#settings.controller1.enabled] // Empty (0) for a switched off controller
    If %v_cs%>0
      ControllerDisable,1
      Let,c1off,1
    Endif
    Let,c2off,0
    Let,cs,0
    Let,cs,[data_5#settings.controller2.enabled]+[data_6#settings.controller2.enabled]+[data_9#settings.controller2.enabled]+[data_11#settings.controller2.enabled]+[data_15#settings.controller2.enabled]+[data_16#settings.controller2.enabled] // Empty (0) for a switched off controller
    If %v_cs%>0
      ControllerDisable,2
      Let,c2off,1
    Endif
    Let,c3off,0
    Let,cs,0
    Let,cs,[data_5#settings.controller3.enabled]+[data_6#settings.controller3.enabled]+[data_9#settings.controller3.enabled]+[data_11#settings.controller3.enabled]+[data_15#settings.controller3.enabled]+[data_16#settings.controller3.enabled] // Empty (0) for a switched off controller
    If %v_cs%>0
      ControllerDisable,3
      Let,c3off,1
    Endif
  Endif
Endon

On TaskInit#data_9 Do
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_9
  Endif
Endon

On TaskInit#data_11 Do
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_11
  Endif
Endon

On TaskInit#data_15 Do
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_15
  Endif
Endon

On TaskInit#data_16 Do // 24 h values: their floor is the restored daily value (Rules 4)
  If %bootcause%=1 And %v_sendok%<1
    TaskDisable,data_16
  Endif
Endon

// First NTP sync after the start: request the saved sum values right away. Timer 3 runs at WiFi#Connected, often before NTP,
// and skips the request without a valid time (Rules 4) - otherwise the restore would wait for the next run (5 minutes)
On Time#Initialized Do
  If %v_savesumvalues%>0 And %v_sumsok%<1
    SendToHTTP,webservices.weatherlox.de,80,"getdata.php?mac={urlencode:"%mac%"}#json3"
  Endif
Endon

// Restore sum values after a restart - only values from the current day / week / month / year
// %eventvalue1-9% = solarradiation_daily, sunduration_daily, windgust_maxdaily, windrun_daily, rain_daily, rain_weekly, rain_monthly, rain_yearly, lightning_daily
// %eventvalue10% = timestamp of the last save (unixtime UTC, set by the server), "null" if there is no saved data
// Only the variables are set here, the task values follow in releasesums
On restoresums Do
  Let,savedts,0
  If %v_sumsok%<1 // Only once per start: after a power loss a second restore would add the saved values again
    Let,savedts,%eventvalue10% // Fails for "null" and savedts stays 0 (a string comparison with null would be true)
  Endif
  Let,sumsok,1 // Restore done (also if there is no saved data), saving to the server and sending may start (Rules 4)
  If %v_savedts%>0
    Let,daystart,%unixtime%-%syssec_d% // Today 0:00 local time
    Let,weekstart,%v_daystart%-((%sysweekday%+5)%7)*86400 // Monday 0:00 (%sysweekday%: 1=Sunday, 2=Monday)
    Let,monthstart,%v_daystart%-(%sysday%-1)*86400 // 1st of month 0:00
    Let,yearstart,(((%sysyear%-1970)*365+((%sysyear%-1969-((%sysyear%-1969)%4))/4))*86400)-%systzoffset_s% // 1st January 0:00
    // Current value rb = max(variable, task value): after a warm restart the task value is kept in the RTC memory and the variable may still be 0,
    // after a power loss the task is blocked (empty = 0) and the variable counts since the start.
    // Power loss (%bootcause%=1, only set if the RTC memory was empty): saved + current. Otherwise the saved value is older: the bigger value.
    // Both in one formula: (rk*(a+b)+rm*|a-b|)/2 with rk=2, rm=0 (sum) or rk=1, rm=1 (max)
    // Brackets around abs() because of an ESPEasy bug (operator before and after a function)
    Let,rk,1
    Let,rm,1
    If %bootcause%=1
      Let,rk,2
      Let,rm,0
    Endif
    If %v_savedts%>=%v_daystart%
      Let,rb,(%v_solarradiation_daily%+[data_11#solarradiation_daily]+(abs(%v_solarradiation_daily%-[data_11#solarradiation_daily])))/2
      Let,solarradiation_daily,(%v_rk%*(%eventvalue1%+%v_rb%)+%v_rm%*(abs(%eventvalue1%-%v_rb%)))/2
      Let,rb,(%v_sunduration_daily%+[data_11#sunduration_daily]+(abs(%v_sunduration_daily%-[data_11#sunduration_daily])))/2
      Let,sunduration_daily,(%v_rk%*(%eventvalue2%+%v_rb%)+%v_rm%*(abs(%eventvalue2%-%v_rb%)))/2
      Let,rb,(%v_windrun_daily%+[data_9#windrun_daily]+(abs(%v_windrun_daily%-[data_9#windrun_daily])))/2
      Let,windrun_daily,(%v_rk%*(%eventvalue4%+%v_rb%)+%v_rm%*(abs(%eventvalue4%-%v_rb%)))/2
      Let,rb,(%v_rain_daily%+[data_6#rain_daily]+(abs(%v_rain_daily%-[data_6#rain_daily])))/2
      Let,rain_daily,(%v_rk%*(%eventvalue5%+%v_rb%)+%v_rm%*(abs(%eventvalue5%-%v_rb%)))/2
      Let,rb,(%v_lightning_daily%+[data_15#lightning_daily]+(abs(%v_lightning_daily%-[data_15#lightning_daily])))/2
      Let,lightning_daily,(%v_rk%*(%eventvalue9%+%v_rb%)+%v_rm%*(abs(%eventvalue9%-%v_rb%)))/2
      Let,rb,(%v_windgust_maxdaily%+[data_9#windgust_maxdaily]+(abs(%v_windgust_maxdaily%-[data_9#windgust_maxdaily])))/2 // Maximum: always the bigger value
      Let,windgust_maxdaily,(%eventvalue3%+%v_rb%+(abs(%eventvalue3%-%v_rb%)))/2
    Endif
    If %v_savedts%>=%v_weekstart%
      Let,rb,(%v_rain_weekly%+[data_6#rain_weekly]+(abs(%v_rain_weekly%-[data_6#rain_weekly])))/2
      Let,rain_weekly,(%v_rk%*(%eventvalue6%+%v_rb%)+%v_rm%*(abs(%eventvalue6%-%v_rb%)))/2
    Endif
    If %v_savedts%>=%v_monthstart%
      Let,rb,(%v_rain_monthly%+[data_6#rain_monthly]+(abs(%v_rain_monthly%-[data_6#rain_monthly])))/2
      Let,rain_monthly,(%v_rk%*(%eventvalue7%+%v_rb%)+%v_rm%*(abs(%eventvalue7%-%v_rb%)))/2
    Endif
    If %v_savedts%>=%v_yearstart%
      Let,rb,(%v_rain_yearly%+[data_6#rain_yearly]+(abs(%v_rain_yearly%-[data_6#rain_yearly])))/2
      Let,rain_yearly,(%v_rk%*(%eventvalue8%+%v_rb%)+%v_rm%*(abs(%eventvalue8%-%v_rb%)))/2
    Endif
  Endif
Endon

// Release the blocked tasks and set the task values from the variables (called once per start by Rules 4, Timer 2)
// Before, each variable takes the task value if that is bigger (warm restart without restore: the variable may still be 0)
On releasesums Do
  Let,sendok,1 // Also stops TaskInit from blocking again
  TaskEnable,data_5
  TaskEnable,data_6
  TaskEnable,data_9
  TaskEnable,data_11
  TaskEnable,data_15
  TaskEnable,data_16
  Let,rb,(%v_rain_event%+[data_5#rain_event]+(abs(%v_rain_event%-[data_5#rain_event])))/2
  Let,rain_event,%v_rb%
  TaskValueSet data_5,rain_event,%v_rain_event%
  Let,rb,(%v_rain_hourly%+[data_5#rain_hourly]+(abs(%v_rain_hourly%-[data_5#rain_hourly])))/2
  Let,rain_hourly,%v_rb%
  TaskValueSet data_5,rain_hourly,%v_rain_hourly%
  Let,rb,(%v_solarradiation_daily%+[data_11#solarradiation_daily]+(abs(%v_solarradiation_daily%-[data_11#solarradiation_daily])))/2
  Let,solarradiation_daily,%v_rb%
  TaskValueSet data_11,solarradiation_daily,%v_solarradiation_daily%
  Let,rb,(%v_sunduration_daily%+[data_11#sunduration_daily]+(abs(%v_sunduration_daily%-[data_11#sunduration_daily])))/2
  Let,sunduration_daily,%v_rb%
  TaskValueSet data_11,sunduration_daily,%v_sunduration_daily%
  Let,rb,(%v_windrun_daily%+[data_9#windrun_daily]+(abs(%v_windrun_daily%-[data_9#windrun_daily])))/2
  Let,windrun_daily,%v_rb%
  TaskValueSet data_9,windrun_daily,%v_windrun_daily%
  Let,rb,(%v_rain_daily%+[data_6#rain_daily]+(abs(%v_rain_daily%-[data_6#rain_daily])))/2
  Let,rain_daily,%v_rb%
  TaskValueSet data_6,rain_daily,%v_rain_daily%
  Let,rb,(%v_lightning_daily%+[data_15#lightning_daily]+(abs(%v_lightning_daily%-[data_15#lightning_daily])))/2
  Let,lightning_daily,%v_rb%
  TaskValueSet data_15,lightning_daily,%v_lightning_daily%
  Let,rb,(%v_rain_weekly%+[data_6#rain_weekly]+(abs(%v_rain_weekly%-[data_6#rain_weekly])))/2
  Let,rain_weekly,%v_rb%
  TaskValueSet data_6,rain_weekly,%v_rain_weekly%
  Let,rb,(%v_rain_monthly%+[data_6#rain_monthly]+(abs(%v_rain_monthly%-[data_6#rain_monthly])))/2
  Let,rain_monthly,%v_rb%
  TaskValueSet data_6,rain_monthly,%v_rain_monthly%
  Let,rb,(%v_rain_yearly%+[data_6#rain_yearly]+(abs(%v_rain_yearly%-[data_6#rain_yearly])))/2
  Let,rain_yearly,%v_rb%
  TaskValueSet data_6,rain_yearly,%v_rain_yearly%
  Let,rb,(%v_windgust_maxdaily%+[data_9#windgust_maxdaily]+(abs(%v_windgust_maxdaily%-[data_9#windgust_maxdaily])))/2
  Let,windgust_maxdaily,%v_rb%
  TaskValueSet data_9,windgust_maxdaily,%v_windgust_maxdaily%
  // Switch the controllers blocked by blockcontrollers on again, after the task values are set
  If %v_c1off%=1
    ControllerEnable,1
  Endif
  If %v_c2off%=1
    ControllerEnable,2
  Endif
  If %v_c3off%=1
    ControllerEnable,3
  Endif
  Let,c1off,0
  Let,c2off,0
  Let,c3off,0
Endon

// Parse data from JSON sources
On JsonReply* Do
  If %eventpar%=1 // Station height for pressure calculations
    Let,stationheight,%eventvalue1%
  Elseif %eventpar%=2 // Passkey for Ecowitt
    LetStr,ecowittpasskey,%eventvalue1%
    Let,passkeyok,1
  Elseif %eventpar%=3 // Saved sum values from server (order as in json.keys)
    Event,restoresums=%eventvalue1%,%eventvalue2%,%eventvalue3%,%eventvalue4%,%eventvalue5%,%eventvalue6%,%eventvalue7%,%eventvalue8%,%eventvalue9%,%eventvalue10%
  Endif
Endon
