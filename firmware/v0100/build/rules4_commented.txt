// Version of Rules 4 for the automatic update (Rules 1) - must match the version in the first line
On version4 Do

Endon
 
// 5sec Tasks / Events
On Rules#Timer=2 Do
  // Release the sum values for the controllers once per start: after the restore (sumsok, Rules 2), at the latest after 15 minutes
  // (server or NTP not reachable - then the restore may still come later and the sums jump)
  If %v_sendok%<1
    If %v_sumsok%=1 Or %uptime%>15
      Let,sendok,1
      Event,releasesums
    Endif
  Endif
  // Resetting Rainrate if no pulse the last 600s
  If %unixtime%>(%v_rainratelast%+600) And [data_5#rainstate]=0
      TaskValueSet data_5,rainrate,0
  Endif
  // No pulse for 600 s, but the rain sensor still detects rain: back to the sensor estimate
  If %unixtime%>(%v_rainratelast%+600) And [data_5#rainstate]=1 And [data_5#rainrate]>0.5
    TaskValueSet data_5,rainrate,0.5
  Endif
  // Rain sensor only (rainrate 0.5 mm/h): credit the rain in advance (0.5/720 mm per 5 s), at most one bucket.
  // The next bucket pulse adds only the rest of the bucket (Rules 3, input_1)
  If [data_5#rainrate]=0.5 And %v_rainest%<(%v_rainbucket%-0.001)
    Let,rainadd,0.5/720
    Let,rainest,%v_rainest%+%v_rainadd%
    Event,addrain=%v_rainadd%
  Endif
  // Resetting Rain Event
  If %unixtime%>(%v_rainratelast%+3600) And [data_5#rainstate]=0
    TaskValueSet data_5,rain_event,0
    Let,rain_event,0
  Endif
  // Prepare and send data for Ecowitt if enabled
  // Only after the sum values are valid (sendok, Rules 2 releasesums): before, the sums are 0 after a power loss and the jump to the restored
  // values would count as rain in some receivers (WeeWX interceptor). Until then the blocks stay empty and the body is empty.
  If [data_1#settings.controller3.enabled]>0 And %v_sendok%=1
    // %unixtime% is UTC and %c_ts2date% does not apply a time zone. The argument is read as float (no calculation!), so the time is up to 64 s off
    // Format like an Ecowitt gateway: date+time without percent encoding ("2026-10-21+09:38:08"),
    // because some receivers (FOSHKplugin, WeeWX interceptor) only decode an uppercase "%3A" ({urlencode} writes "%3a")
    LetStr,ecowittdateutc,"{substring:0:10:"%c_ts2date%(%unixtime%)"}+{substring:11:19:"%c_ts2date%(%unixtime%)"}"
    Let,ecowitttempf,%c_c2f%([data_1#temperature])
    Let,ecowittbaromabsin,[data_3#pressure_abs]*0.02953
    Let,ecowittbaromrelin,[data_3#pressure_rel]*0.02953
    Let,ecowittrainratein,[data_5#rainrate]/25.4
    Let,ecowitteventrainin,[data_5#rain_event]/25.4
    Let,ecowitthourlyrainin,[data_5#rain_hourly]/25.4
    Let,ecowittdailyrainin,[data_6#rain_daily]/25.4
    Let,ecowittweeklyrainin,[data_6#rain_weekly]/25.4
    Let,ecowittmonthlyrainin,[data_6#rain_monthly]/25.4
    Let,ecowittyearlyrainin,[data_6#rain_yearly]/25.4
    Let,ecowittwindspeedmph,[data_7#windspeed]*0.621371
    Let,ecowittwindgustmph,[data_7#windgust_max2m]*0.621371
    Let,ecowittwindspdmph_avg2m,[data_7#windspeed_avg2m]*0.621371
    Let,ecowittwindspdmph_avg10m,[data_7#windspeed_avg10m]*0.621371
    Let,ecowittlast24hrainin,[data_16#rain_24h]/25.4
    Let,ecowittwindgustmph_max24h,[data_16#windgust_max24h]*0.621371
    // sunhours and windgustmph_max24h have no Ecowitt key: names as in FOSHKplugin (with sunhours it does not calculate its own value)
    LetStr,ecowittblock1,"mac=[str#ecowittmac]&PASSKEY=[str#ecowittpasskey]&dateutc=[str#ecowittdateutc]&stationtype=[str#ecowittstationtype]&model=[str#ecowittmodel]&"
    LetStr,ecowittblock2,"twilight=[data_10#twilight]&brightness=[data_10#illuminance]&uv=[data_10#uvi]&solarradiation=[data_10#solarradiation]&sunhours=[data_11#sunduration_daily]&"
    LetStr,ecowittblock3,"tempf=%v_ecowitttempf%&baromabsin=%v_ecowittbaromabsin%&baromrelin=%v_ecowittbaromrelin%&humidity=[data_1#humidity]&"
    LetStr,ecowittblock4,"srain_piezo=[data_5#rainstate]&rainratein=%v_ecowittrainratein%&eventrainin=%v_ecowitteventrainin%&last24hrainin=%v_ecowittlast24hrainin%&"
    LetStr,ecowittblock5,"hourlyrainin=%v_ecowitthourlyrainin%&dailyrainin=%v_ecowittdailyrainin%&weeklyrainin=%v_ecowittweeklyrainin%&monthlyrainin=%v_ecowittmonthlyrainin%&yearlyrainin=%v_ecowittyearlyrainin%&"
    LetStr,ecowittblock6,"lightning_time=[data_15#lightning_time]&lightning_num=[data_15#lightning_daily]&lightning=[data_15#lightning]&"
    LetStr,ecowittblock7,"winddir=[data_8#winddir]&winddir_avg2m=[data_8#winddir_avg2m]&winddir_avg10m=[data_8#winddir_avg10m]&windgustmph_max24h=%v_ecowittwindgustmph_max24h%&"
    // Last block without a trailing "&" (FOSHKplugin drops a body that ends with "&")
    LetStr,ecowittblock8,"windspeedmph=%v_ecowittwindspeedmph%&windgustmph=%v_ecowittwindgustmph%&windspdmph_avg2m=%v_ecowittwindspdmph_avg2m%&windspdmph_avg10m=%v_ecowittwindspdmph_avg10m%" 
  Endif
Endon

// Minutely Tasks / Events
On Clock#Time=All,**:** Do
  // ESPEasy reads internal_3 once at the start: the task values from the RTC memory (last hour after a warm restart)
  // would count a second time in the hourly statistics. Clear the samples once per start, before the first hourly sample.
  If %v_i3init%<1
    internal_3.clearsamples
    Let,i3init,1
  Endif
  // Hourly Tasks / Events
  If %sysmin%=0
    // Hourly statistics for the 24 h values: rain and highest gust of the hour that just ended.
    // internal_3 has Stats enabled for all values and runs once per hour, so sample1 is the last full hour.
    // While data_5/data_16 are blocked after a power loss (Rules 2), [task#value] is empty: take the variable and 0 instead.
    Let,rb,%v_rain_hourly%
    Let,gb,0
    If %bootcause%!=1 Or %v_sendok%=1
      Let,rb,(%v_rain_hourly%+[data_5#rain_hourly]+(abs(%v_rain_hourly%-[data_5#rain_hourly])))/2 // After a warm restart the variable may still be 0
      Let,gb,[data_16#windgust_maxhourly]
    Endif
    TaskValueSet internal_3,int_random,%c_random%(0, 1) // Force the Statistics to be updated (e.g. several dry hours in a row)
    TaskValueSet internal_3,int_windgust_maxhourly,%v_gb%
    TaskValueSetAndRun internal_3,int_rain_hourly,%v_rb%
    TaskValueSet data_16,windgust_maxhourly,0
    TaskValueSet data_5,rain_hourly,0
    Let,rain_hourly,0
  Endif
  // 24 h values: the last 23 full hours (internal_3) plus the current hour, so 23 to 24 hours.
  // At least the daily value: it covers at most 24 hours and is restored after a power loss, while the Stats are empty after every restart.
  // ESPEasy has no sum for Stats: sum = average of the last n samples * n
  If %v_sendok%=1
    Let,r24,[data_5#rain_hourly]
    Let,g24,[data_16#windgust_maxhourly]
    Let,n24,[internal_3#int_rain_hourly.size]
    If %v_n24%>0
      If %v_n24%>23
        Let,n24,23
      Endif
      Let,r24,%v_r24%+[internal_3#int_rain_hourly.avg23]*%v_n24%
      Let,gb,[internal_3#int_windgust_maxhourly.max23]
      Let,g24,(%v_g24%+%v_gb%+(abs(%v_g24%-%v_gb%)))/2
    Endif
    Let,r24,(%v_r24%+[data_6#rain_daily]+(abs(%v_r24%-[data_6#rain_daily])))/2
    Let,g24,(%v_g24%+[data_9#windgust_maxdaily]+(abs(%v_g24%-[data_9#windgust_maxdaily])))/2
    TaskValueSet data_16,rain_24h,%v_r24%
    TaskValueSet data_16,windgust_max24h,%v_g24%
  Endif
  // Daily, Weekly, Monthly, Yearly Tasks / Events
  If %syssec_d%<59
    Let,solarradiation_daily,0
    TaskValueSet data_11,solarradiation_daily,0
    Let,sunduration_daily,0
    TaskValueSet data_11,sunduration_daily,0
    TaskValueSet data_9,windgust_maxdaily,0
    TaskValueSet data_9,windrun_daily,0
    Let,windrun_daily,0
    TaskValueSet data_6,rain_daily,0
    Let,rain_daily,0
    TaskValueSet data_15,lightning_daily,0
    Let,lightning_daily,0
    If %sysweekday%=2 // Weekly
      TaskValueSet data_6,rain_weekly,0
      Let,rain_weekly,0
    Endif
    If %sysday%=1 // Monthly
      TaskValueSet data_6,rain_monthly,0
      Let,rain_monthly,0
    Endif
    If %sysday%=1 And %sysmonth%=1 // Yearly
      TaskValueSet data_6,rain_yearly,0
      Let,rain_yearly,0
    Endif
  Endif
  // Minutely Tasks
  // Calculate pressure trends
  // See: https://www.nxp.com/docs/en/application-note/AN3914.pdf
  TaskValueSet internal_2,int_random,%c_random%(0, 1) // Force the Statistics to be updated
  TaskValueSetAndRun internal_2,int_pressure,[data_3#pressure_abs.avg60] // Average of the last 5 minutes (data_3 has Stats enabled, interval 5 s), independent of the pressure sensor
  // sample1 is the newest value (one per minute), so sample61 is 60 minutes ago and sample181 is 180 minutes ago
  If [internal_2#int_pressure.sample]>60
    TaskValueSet data_3,pressurechange1,[internal_2#int_pressure]-[internal_2#int_pressure.sample61]
    If [data_3#pressurechange1]>0.7
      TaskValueSet data_4,pressuretrend1,2
    Elseif [data_3#pressurechange1]>0.2
      TaskValueSet data_4,pressuretrend1,1
    Elseif [data_3#pressurechange1]<-0.7
      TaskValueSet data_4,pressuretrend1,-2
    Elseif [data_3#pressurechange1]<-0.2
      TaskValueSet data_4,pressuretrend1,-1
    Else
      TaskValueSet data_4,pressuretrend1,0
    Endif
  Endif
  If [internal_2#int_pressure.sample]>180
    TaskValueSet data_4,pressurechange3,[internal_2#int_pressure]-[internal_2#int_pressure.sample181]
    If [data_4#pressurechange3]>2
      TaskValueSet data_4,pressuretrend3,2
    Elseif [data_4#pressurechange3]>0.7
      TaskValueSet data_4,pressuretrend3,1
    Elseif [data_4#pressurechange3]<-2
      TaskValueSet data_4,pressuretrend3,-2
    Elseif [data_4#pressurechange3]<-0.7
      TaskValueSet data_4,pressuretrend3,-1
    Else
      TaskValueSet data_4,pressuretrend3,0
    Endif
  Endif
  Event,calcsun=%latitude%,%longitude%,%unixtime% // Sun position and threshold for sunshine duration
Endon

// Sun position and threshold (seuil) for sunshine duration - same algorithm as FOSHKplugin theoreticalSun
// %eventvalue1% = latitude, %eventvalue2% = longitude, %eventvalue3% = unixtime (UTC)
// See: https://github.com/Jterrettaz/sunduration
// See: http://meteo-sciez.fr/O1_07_Vuerich_Sunshine_Duration.pdf
// See: https://library.wmo.int/viewer/68695/?offset=#page=344&viewer=picture&o=bookmark&n=0&q=
// ESPEasy bug: a function with an operator before and after it needs brackets around it, e.g. 2*(sin_d(x))+1
On calcsun Do
  If %eventvalue1%=0 And %eventvalue2%=0 // No coordinates set: fixed WMO threshold 120 W/m2 like FOSHKplugin
    Let,seuil,120
  Else
    Let,utcsec,fmod(%eventvalue3%:86400) // Seconds since midnight UTC
    Let,dayofyear,((%eventvalue3%-%v_utcsec%)/86400)-((%sysyear%-1970)*365)-((%sysyear%-1969-((%sysyear%-1969)%4))/4)+1 // Days since 1970 minus days until 1st January (incl. leap days, valid 1901-2099)
    Let,thetasd,360*%v_dayofyear%/365
    Let,equatemps,0.0172+0.4281*(cos_d(%v_thetasd%))-7.3515*(sin_d(%v_thetasd%))-3.3495*(cos_d(2*%v_thetasd%))-9.3619*(sin_d(2*%v_thetasd%)) // Equation of time (min)
    Let,declinaison,asin_d(0.006918-0.399912*(cos_d(%v_thetasd%))+0.070257*(sin_d(%v_thetasd%))-0.006758*(cos_d(2*%v_thetasd%))+0.000908*(sin_d(2*%v_thetasd%)))
    Let,anglehoraire,((%v_utcsec%/60+%eventvalue2%*4+%v_equatemps%)/60-12)*15 // Hour angle from solar time: minutes UTC + longitude correction + equation of time
    Let,hauteursoleil,asin_d((sin_d(%eventvalue1%))*(sin_d(%v_declinaison%))+(cos_d(%eventvalue1%))*(cos_d(%v_declinaison%))*(cos_d(%v_anglehoraire%)))
    TaskValueSet data_11,sunelevation,%v_hauteursoleil%
    Let,suny,-1*(sin_d(%v_anglehoraire%))
    Let,sunx,((tan_d(%v_declinaison%))*(cos_d(%eventvalue1%)))-((sin_d(%eventvalue1%))*(cos_d(%v_anglehoraire%)))
    TaskValueSet data_11,sundirection,fmod((atan2_d(%v_suny%:%v_sunx%)+360):360) // Azimuth 0=N, 90=E, 180=S, 270=W
    If %v_hauteursoleil%>3
      Let,seuil,(0.73+%v_bcoeff%*(cos_d(%v_thetasd%)))*1080*((sin_d(%v_hauteursoleil%))^1.25)*%v_globcoeff%
    Else
      Let,seuil,0
    Endif
  Endif
Endon

// 5min Tasks / Events
On Rules#Timer=3 Do
  // Try to enable tasks for I2C sensors again in case they were lost (Rules 2)
  Event,EnableSensors
  // Requests to web services that are repeated until they got an answer (first run right after WiFi#Connected, Rules 2)
  // Station height from Open Meteo, if not set manually in Rules 1
  If %v_stationheight%=0
    SendToHTTP,api.open-meteo.com,80,"v1/elevation?latitude=%latitude%&longitude=%longitude%#json1"
  Endif
  // Passkey for the Ecowitt format, if enabled - ESPEasy cannot calculate MD5 checksums
  If [data_1#settings.controller3.enabled]>0 And %v_passkeyok%<1
    SendToHTTP,webservices.weatherlox.de,80,"ecowittpasskey.php?mac={urlencode:"%mac%"}#json2"
  Endif
  // Sum values: restore them from the weatherlox webserver after a restart, then save them every 5 minutes (for a power loss)
  If %v_savesumvalues%>0
    // No timeout: as long as the server does not answer, nothing is saved (the saved values stay for the restore)
    If %v_sumsok%<1
      // Do not save before the restore, it would overwrite the saved values
      // Restore only with a valid time (after 1.1.2026): without NTP the clock starts at 1.1.1970 after a power loss
      // and every saved value would count as one of today (Rules 2, restoresums)
      If %unixtime%>1767225600
        SendToHTTP,webservices.weatherlox.de,80,"getdata.php?mac={urlencode:"%mac%"}#json3"
      Endif
    Else
      SendToHTTP,webservices.weatherlox.de,80,"savedata.php?mac={urlencode:"%mac%"}&solarradiation_daily=[data_11#solarradiation_daily]&sunduration_daily=[data_11#sunduration_daily]&windgust_maxdaily=[data_9#windgust_maxdaily]&windrun_daily=[data_9#windrun_daily]&rain_daily=[data_6#rain_daily]&rain_weekly=[data_6#rain_weekly]&rain_monthly=[data_6#rain_monthly]&rain_yearly=[data_6#rain_yearly]&lightning_daily=[data_15#lightning_daily]"
    Endif
  Else
    Let,sumsok,1 // Sum values are not saved: nothing to restore, Ecowitt may send (Rules 4, Timer 2)
  Endif
Endon

// Quarterly Tasks / Events
On Rules#Timer=1 Do
  // Get weather data from Open Meteo if enabled
  If %v_getopenmeteo%>0
    SendToHTTP,api.open-meteo.com,80,"v1/forecast?latitude=%latitude%&longitude=%longitude%&current=precipitation_probability,visibility,evapotranspiration,et0_fao_evapotranspiration,weather_code,cloud_cover"
  Endif
Endon
